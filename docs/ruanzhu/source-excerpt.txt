/// True when a shortcut, notification payload or `zhuaji://` route should
/// open the camera. The path may arrive as `/scan` or nested in `route`.
bool routeSelectsScan(String route) {
  final trimmed = route.trim();
  final uri = Uri.tryParse(trimmed);
  if (uri == null) return trimmed == '/scan';
  if (uri.path == '/scan') return true;
  final nested = uri.queryParameters['route'];
  if (nested == null || nested == trimmed) return false;
  return routeSelectsScan(nested);
}

/// External entries. A scan target selects 今天 first, then the camera, so
/// closing the camera lands on the page that holds the scan button.
void openAppRoute(GoRouter router, String route) {
  if (routeSelectsScan(route)) {
    router.go('/home');
    router.push('/scan');
    return;
  }
  router.push(route);
}

enum AppEnv { local, staging, prod }

/// API base URL for the current build.
///
/// Switch with `--dart-define=API_ENV=local|staging|prod`.
/// A full override wins: `--dart-define=API_BASE_URL=http://10.0.2.2:8094`.
class AppConfig {
  const AppConfig({this.env = AppEnv.local, this.baseUrlOverride});

  factory AppConfig.fromEnvironment() {

花爪软件 V2.18.0                        第 8 页
    const envName = String.fromEnvironment('API_ENV', defaultValue: 'local');
    const override = String.fromEnvironment('API_BASE_URL');
    return AppConfig(
      env: parseAppEnv(envName),
      baseUrlOverride: override.isEmpty ? null : override,
    );
  }

  final AppEnv env;
  final String? baseUrlOverride;

  String get apiBaseUrl {
    final override = baseUrlOverride;
    if (override != null && override.isNotEmpty) {
      return override.endsWith('/')
          ? override.substring(0, override.length - 1)
          : override;
    }
    return switch (env) {
      AppEnv.local => 'http://10.0.2.2:8094',
      AppEnv.staging => 'https://staging-api.zhuaji.local',
      AppEnv.prod => 'https://api.zhuaji.local',
    };
  }
}

AppEnv parseAppEnv(String raw) {
  return switch (raw) {
    'staging' => AppEnv.staging,
    'prod' => AppEnv.prod,
    _ => AppEnv.local,
  };
}

String ageLabel(DateTime? birthday, DateTime now) {
  if (birthday == null) return '年龄未填';
  var months = (now.year - birthday.year) * 12 + now.month - birthday.month;
  if (now.day < birthday.day) months -= 1;
  if (months < 1) return '未满 1 个月';
  if (months < 12) return '$months 个月';
  final years = months ~/ 12;
  final rest = months % 12;
  if (rest == 0) return '$years 岁';
  return '$years 岁 $rest 个月';
}

/// One key per product code: a GS1 string and the plain EAN-13 printed under
/// it both reduce to the same digits, so either scan finds the memory.
String barcodeKey(String raw) {
  final code = raw.trim();
  if (code.isEmpty) return '';
  final gtin = code.length > 14 ? const Gs1Parser().parse(code).gtin : null;
  final digits = (gtin ?? code).replaceAll(RegExp(r'\s'), '');
  if (RegExp(r'^\d+$').hasMatch(digits)) {
    final trimmed = digits.replaceFirst(RegExp(r'^0+'), '');
    return trimmed.isEmpty ? '0' : trimmed;
  }
  return digits;
}
